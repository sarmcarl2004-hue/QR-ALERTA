/*
 * =========================================================================
 *              QR-ALERTA CHIMBOTE - FIRMWARE DEL POSTE INTELIGENTE (ESP32)
 * =========================================================================
 * Curso: Creatividad Empresarial (UTP Chimbote - 2026)
 * Protocolo: MQTT sobre WiFi (Suscripción a base de datos del paradero)
 * Componentes requeridos: ESP32 NodeMCU, Módulo Relé 5V, Sirena 12V, Fuente 12V.
 */

#include &lt;WiFi.h&gt;
#include &lt;PubSubClient.h&gt; // Instalar desde el Gestor de Librerías

// --- CONFIGURACIÓN DE RED ---
const char* ssid = "TU_WIFI_SSID";          // Nombre de tu red WiFi
const char* password = "TU_WIFI_PASSWORD";  // Clave de tu red WiFi

// --- CONFIGURACIÓN DEL BROKER MQTT (GRATUITO Y PÚBLICO) ---
const char* mqtt_server = "broker.hivemq.com";
const int mqtt_port = 1883;
// El tema debe coincidir exactamente con el canal de tu placa
const char* topic_sirena = "utp/chimbote/alerta/poste042"; 

// --- ASIGNACIÓN DE PINES FÍSICOS (GPIO) ---
const int RELAY_PIN = 14;  // Pin D14 en ESP32 para controlar el relé de la Sirena
const int LED_PIN = 2;     // Luz LED integrada para pruebas rápidas de conexión

WiFiClient espClient;
PubSubClient client(espClient);

void setup() {
  Serial.begin(115200);
  
  // Configurar Pines
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // Mantener sirena apagada al inicio
  digitalWrite(LED_PIN, LOW);

  // Inicializar WiFi y MQTT
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
}

void setup_wifi() {
  delay(10);
  Serial.println();
  Serial.print("Conectando a ");
  Serial.println(ssid);

  WiFi.begin(ssid, password);

    digitalWrite(LED_PIN, !digitalRead(LED_PIN));

  Serial.println("");

  Serial.print("Dirección IP asignada: ");

// Escuchador de Mensajes en tiempo real
void callback(char* topic, byte* payload, unsigned int length) {
  Serial.print("Mensaje recibido en el tema: [");
  Serial.print(topic);
  Serial.print("] ");

  String messageTemp;
  for (int i = 0; i < length; i++) {
    messageTemp += (char)payload[i];
  }
  Serial.println(messageTemp);

  // Evaluar orden de activación
  if (messageTemp == "ACTIVAR" || messageTemp == "ON") {
    Serial.println("📢 ORDEN RECIBIDA: ¡ACTIVANDO SIRENA Y ESTROBOS EN EL POSTE!");
    digitalWrite(RELAY_PIN, HIGH); // Cierra el relé (Suena la sirena de 12V)
    
    // Simular un tiempo de apagado automático de seguridad tras 20 segundos
    delay(20000); 
    
    Serial.println("📢 TIEMPO LÍMITE: Apagando sirena para evitar saturación de ruido.");
    digitalWrite(RELAY_PIN, LOW);  // Abre el relé (Se silencia la sirena)
  } 
  else if (messageTemp == "SILENCIAR" || messageTemp == "OFF") {
    Serial.println("📢 ORDEN RECIBIDA: Silenciando sirena de forma remota.");
    digitalWrite(RELAY_PIN, LOW);
  }
}

void reconnect() {
  // Bucle de reconexión automática MQTT
  while (!client.connected()) {
    Serial.print("Intentando conectar al Broker MQTT...");
    // Generar un ID único para evitar colisiones
    String clientId = "ESP32_Chimbote_Poste_";
    clientId += String(random(0xffff), HEX);
    
    if (client.connect(clientId.c_str())) {
      Serial.println("Conectado exitosamente al Broker!");
      // Volver a suscribirse al canal
      client.subscribe(topic_sirena);
    } else {
      Serial.print("Error de conexión, rc=");
      Serial.print(client.state());
      Serial.println(" Reintentando en 5 segundos...");
      delay(5000);
    }
  }
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop(); // Mantiene el hilo de escucha activo

        window.cambiarPestana = function(pestanaId) {
            // Secciones a controlar
            const sections = {
                simulador: document.getElementById('section-simulador'),
                codigo: document.getElementById('section-codigo'),
                esquema: document.getElementById('section-esquema'),
                despliegue: document.getElementById('section-despliegue')
            };

            // Botones de pestañas
            const tabs = {
                simulador: document.getElementById('tab-simulador'),
                codigo: document.getElementById('tab-codigo'),
                esquema: document.getElementById('tab-esquema'),
                despliegue: document.getElementById('tab-despliegue')
            };

            // Apagar y resetear todas las secciones
            Object.keys(sections).forEach(key => {
                if (sections[key]) {
                    sections[key].classList.add('hidden');
                }
                if (tabs[key]) {
                    tabs[key].className = "flex-1 min-w-[120px] text-xs font-bold py-3 px-4 rounded-xl transition duration-200 flex items-center justify-center gap-2 text-slate-400 hover:text-slate-200 hover:bg-slate-800/50";
                }
            });

            // Activar la pestaña elegida
            if (sections[pestanaId]) {
                sections[pestanaId].classList.remove('hidden');
            }
            if (tabs[pestanaId]) {
                tabs[pestanaId].className = "flex-1 min-w-[120px] text-xs font-bold py-3 px-4 rounded-xl transition duration-200 flex items-center justify-center gap-2 bg-blue-600 text-white shadow-md";
            }
        }

        // Herramienta de copiado seguro compatible con sandboxes de navegadores
        window.copiarAlPortapapeles = function(elementId) {
            const preElement = document.getElementById(elementId);
            if (!preElement) return;
            
            const tempTextArea = document.createElement("textarea");
            tempTextArea.value = preElement.innerText;
            document.body.appendChild(tempTextArea);
            tempTextArea.select();
            
            try {
                document.execCommand('copy');
                // Alerta elegante simulada temporalmente sin usar alert()
                const originalText = document.getElementById('tab-' + elementId.split('-')[1]).innerHTML;
                const tabBtn = document.getElementById('tab-' + elementId.split('-')[1]);
                tabBtn.innerHTML = "<span>¡Copiado con éxito!</span>";
                setTimeout(() => {
                    tabBtn.innerHTML = originalText;
                }, 1500);
            } catch (err) {
                console.warn("No se pudo realizar el copiado automático", err);
            }
            
            document.body.removeChild(tempTextArea);
        }
